# CatPred EMULaToR bench commands

# Use the requested environment and GPU.
# Stop any long bench command with Ctrl+C once. The wrappers terminate active
# worker/train child processes and keep already-written caches/checkpoints.
cd /home/adhil/github/CatPred
export TORCH_FORCE_NO_WEIGHTS_ONLY_LOAD=1
export CUDA_VISIBLE_DEVICES=1

# List or stop stuck bench processes from this user.
python emulator_bench/kill_bench.py
python emulator_bench/kill_bench.py --yes

# 0) One-time fair-esm isolation for CatPred ESM2.
# This avoids clobbering the global `esm` package used by ESM3.
python -m pip install fair-esm \
  -t /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/embeddings/fair_esm

# 1) One-command default retraining.
# Remove skip flags for any stage you want the launcher to rebuild.
conda run -n mldb python emulator_bench/launch_parallel_retrain.py \
  --gpus 0 \
  --seeds 666 777 888 \
  --value_types kcat \
  --ensemble_size 1 \
  --run_name paper_defaults_single_model \
  --skip_align \
  --skip_embeddings \
  --skip_precompute \
  --num_workers 0 \
  --cpu_threads 2 \
  --interop_threads 1 \
  --esm_mem_cache_max 128 \
  --cuda_empty_cache_interval 100 \
  --max_parallel_per_gpu 6

# 2) Dry-run discovery without writes or training.
conda run -n mldb python emulator_bench/launch_parallel_retrain.py \
  --gpus 1 \
  --dry_run

# 3) Run only one split family/value type for smoke/debug.
conda run -n mldb python emulator_bench/launch_parallel_retrain.py \
  --gpus 1 \
  --value_types kcat \
  --split_groups random_splits_grouped_sequence \
  --seeds 42

# 4) Run alignment only. Original split files are backed up before in-place edits.
conda run -n mldb python emulator_bench/align_structures.py \
  --base_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred \
  --value_types kcat km ki

# 5) Build one-time ESM2 and proGRES caches only.
CUDA_VISIBLE_DEVICES=1 python emulator_bench/cache_embeddings.py \
  --base_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred \
  --cache_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/embeddings \
  --progres_device cuda:0 \
  --esm_gpus 0

# 6) CPU precompute only, using process workers.
conda run -n mldb python emulator_bench/precompute_features.py \
  --base_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred \
  --cache_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/embeddings \
  --feature_root /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/features \
  --seeds 666 777 888 \
  --jobs 5 \
  --max_tasks_per_child 1

# 7) Train one explicit split after caches exist.
CUDA_VISIBLE_DEVICES=1 conda run -n mldb python emulator_bench/train_single_target_tvt.py \
  --train_csv /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/kcat/random_splits_grouped_sequence/train.parquet \
  --val_csv /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/kcat/random_splits_grouped_sequence/val.parquet \
  --test_csv /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/kcat/random_splits_grouped_sequence/test.parquet \
  --out_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred/kcat/random_splits_grouped_sequence/catpred_results/paper_defaults/seed_42 \
  --device cuda:0 \
  --seed 42 \
  --cache_batch_graphs \
  --strict_precompute \
  --require_cached_esm

# 8) Optuna tuning example, separate from default retraining.
CUDA_VISIBLE_DEVICES=1 conda run -n mldb python emulator_bench/tune_optuna.py \
  --base_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred \
  --value_type kcat \
  --split_groups random_splits_grouped_sequence \
  --n_trials 20 \
  --sweep_batch_size \
  --metric RMSE \
  --eval_split val \
  --storage sqlite:////home/adhil/github/EMULaToR/data/processed/baselines/CatPred/optuna_catpred_kcat.db \
  --study_name catpred_kcat_rmse

# 9) Aggregate kcat/km/ki results and write aggregate + missing-result reports.
conda run -n mldb python emulator_bench/aggregate_catpred_results.py \
  --base_dir /home/adhil/github/EMULaToR/data/processed/baselines/CatPred \
  --run_names paper_defaults_single_model \
  --seeds 666 777 888
